    <script>
        let runningTotal = 0;
        let totalCardsDealt = 0;
        let totalDecksInShoe = 8;
        let tableMin = 10;
        let currentSeat = "Seat 1";
        let historyStack = [];

        function saveState() {
            historyStack.push({
                runningTotal,
                totalCardsDealt,
                totalDecksInShoe,
                currentSeat,
                logText: document.getElementById('history-log').innerText
            });
        }

        function recordCardGroup(type) {
            saveState();
            let change = 0;
            let cardsCounted = 1;
            if (type === 'low') { runningTotal += 1; change = +1; }
            else if (type === 'high') { runningTotal -= 1; change = -1; }

            totalCardsDealt += cardsCounted;
            
            let actionName = type === 'low' ? 'Low (+1)' : type === 'high' ? 'High (-1)' : 'Neut (0)';
            updateLog(`${currentSeat}: Recorded ${actionName}`);
            updateDisplay();
        }

        function setSeat(seatNum) {
            currentSeat = typeof seatNum === 'number' ? `Seat ${seatNum}` : seatNum;
            document.querySelectorAll('.seat-btn').forEach(btn => {
                btn.classList.toggle('active', btn.innerText === currentSeat);
            });
            document.getElementById('active-seat').innerText = currentSeat;
        }

        function burnCards() {
            let val = parseInt(document.getElementById('burn-input').value);
            if (isNaN(val) || val <= 0) return;
            saveState();
            totalCardsDealt += val;
            updateLog(`Burned ${val} card(s)`);
            document.getElementById('burn-input').value = '';
            updateDisplay();
        }

        function adjustDeckCount() {
            if (totalDecksInShoe > 1) {
                saveState();
                totalDecksInShoe -= 1;
                updateLog(`Removed 1 deck from calculation`);
                updateDisplay();
            }
        }

        function updateLog(msg) {
            document.getElementById('history-log').innerText = `Last Action: ${msg}`;
        }

        function updateDisplay() {
            document.getElementById('running-total').innerText = runningTotal;
            
            let totalCardsInShoe = totalDecksInShoe * 52;
            let cardsRemaining = Math.max(0, totalCardsInShoe - totalCardsDealt);
            let decksRemaining = Math.max(0.5, (cardsRemaining / 52).toFixed(1));
            
            document.getElementById('decks-remaining').innerText = decksRemaining;
            document.getElementById('cards-left').innerText = cardsRemaining;

            let trueCount = (runningTotal / decksRemaining).toFixed(1);
            document.getElementById('true-count').innerText = trueCount;

            // Recommended Bet calculation
            let recBet = tableMin;
            let tcNum = parseFloat(trueCount);
            if (tcNum >= 2) recBet = tableMin * Math.min(8, Math.floor(tcNum) * 2);
            document.getElementById('recommended-bet').innerText = `$${recBet}`;

            // Insurance Warning
            let banner = document.getElementById('insurance-banner');
            if (tcNum >= 3) {
                banner.style.display = 'block';
                banner.innerText = `🛡️ TRUE COUNT HIGH (+${tcNum}): TAKE INSURANCE`;
                banner.style.background = '#d1fae5';
                banner.style.borderColor = '#10b981';
                banner.style.color = '#065f46';
            } else {
                banner.style.display = 'none';
            }

            calculateDecision();
        }

        function calculateDecision() {
            let hand = document.getElementById('player-hand').value;
            let dealer = parseInt(document.getElementById('dealer-upcard').value);
            let tc = parseFloat((runningTotal / Math.max(0.5, parseFloat(document.getElementById('decks-remaining').innerText))).toFixed(1));

            let move = "HIT";
            let deviationText = "";
            let isSplit = false;

            // --- PAIR STRATEGY & DEVIATIONS ---
            if (hand === 'AA' || hand === '88') { move = "SPLIT"; isSplit = true; }
            else if (hand === '1010') {
                if (dealer === 5 && tc >= 5) { move = "SPLIT"; isSplit = true; deviationText = "I18 #4: Split 10s vs 5 (TC ≥ +5)"; }
                else if (dealer === 6 && tc >= 4) { move = "SPLIT"; isSplit = true; deviationText = "I18 #5: Split 10s vs 6 (TC ≥ +4)"; }
                else { move = "STAND"; }
            }
            else if (hand === '99') {
                if ([7, 10, 11].includes(dealer)) { move = "STAND"; }
                else { move = "SPLIT"; isSplit = true; }
            }
            else if (['77', '33', '22'].includes(hand)) {
                if (dealer >= 2 && dealer <= 7) { move = "SPLIT"; isSplit = true; }
            }
            else if (hand === '66') {
                if (dealer >= 2 && dealer <= 6) { move = "SPLIT"; isSplit = true; }
            }
            else if (hand === '44') {
                if (dealer >= 5 && dealer <= 6) { move = "SPLIT"; isSplit = true; }
            }
            else if (hand === '55') {
                // Treated as hard 10
                move = (dealer <= 9) ? "DOUBLE" : "HIT";
            }
            // --- HARD TOTALS & DEVIATIONS ---
            else if (hand === '16') {
                if (dealer === 10 && tc >= 0) { move = "STAND"; deviationText = "I18 #2: Stand 16 vs 10 (TC ≥ 0)"; }
                else if (dealer === 9 && tc >= 5) { move = "STAND"; deviationText = "I18 #13: Stand 16 vs 9 (TC ≥ +5)"; }
                else if (dealer >= 9) { move = "HIT"; }
                else { move = "STAND"; }
            }
            else if (hand === '15') {
                if (dealer === 10) {
                    if (tc >= 4) { move = "STAND"; deviationText = "I18 #3: Stand 15 vs 10 (TC ≥ +4)"; }
                    else if (tc >= 0) { move = "SURRENDER"; deviationText = "Fab 4: Surrender 15 vs 10 (TC ≥ 0)"; }
                    else { move = "HIT"; }
                } else if (dealer === 9 && tc >= 2) {
                    move = "SURRENDER"; deviationText = "Fab 4: Surrender 15 vs 9 (TC ≥ +2)";
                } else if (dealer === 11 && tc >= 1) {
                    move = "SURRENDER"; deviationText = "Fab 4: Surrender 15 vs Ace (TC ≥ +1)";
                } else {
                    move = (dealer <= 6) ? "STAND" : "HIT";
                }
            }
            else if (hand === '14') {
                if (dealer === 10 && tc >= 3) { move = "SURRENDER"; deviationText = "Fab 4: Surrender 14 vs 10 (TC ≥ +3)"; }
                else { move = (dealer <= 6) ? "STAND" : "HIT"; }
            }
            else if (hand === '13') {
                if (dealer >= 2 && dealer <= 3 && tc <= -2) { move = "HIT"; deviationText = "I18 #18: Hit 13 vs 3 (TC ≤ -2)"; }
                else if (dealer === 2 && tc <= -1) { move = "HIT"; deviationText = "I18 #14: Hit 13 vs 2 (TC ≤ -1)"; }
                else { move = (dealer <= 6) ? "STAND" : "HIT"; }
            }
            else if (hand === '12') {
                if (dealer === 3 && tc >= 2) { move = "STAND"; deviationText = "I18 #7: Stand 12 vs 3 (TC ≥ +2)"; }
                else if (dealer === 2 && tc >= 3) { move = "STAND"; deviationText = "I18 #8: Stand 12 vs 2 (TC ≥ +3)"; }
                else if (dealer === 4 && tc < 0) { move = "HIT"; deviationText = "I18 #15: Hit 12 vs 4 (TC < 0)"; }
                else if (dealer === 5 && tc <= -2) { move = "HIT"; deviationText = "I18 #16: Hit 12 vs 5 (TC ≤ -2)"; }
                else if (dealer === 6 && tc <= -1) { move = "HIT"; deviationText = "I18 #17: Hit 12 vs 6 (TC ≤ -1)"; }
                else { move = (dealer >= 4 && dealer <= 6) ? "STAND" : "HIT"; }
            }
            else if (hand === '11') {
                if (dealer === 11 && tc >= 1) { move = "DOUBLE"; deviationText = "I18 #9: Double 11 vs Ace (TC ≥ +1)"; }
                else { move = "DOUBLE"; }
            }
            else if (hand === '10') {
                if (dealer === 10 && tc >= 4) { move = "DOUBLE"; deviationText = "I18 #6: Double 10 vs 10 (TC ≥ +4)"; }
                else if (dealer === 11 && tc >= 4) { move = "DOUBLE"; deviationText = "I18 #11: Double 10 vs Ace (TC ≥ +4)"; }
                else { move = (dealer <= 9) ? "DOUBLE" : "HIT"; }
            }
            else if (hand === '9') {
                if (dealer === 2 && tc >= 1) { move = "DOUBLE"; deviationText = "I18 #10: Double 9 vs 2 (TC ≥ +1)"; }
                else if (dealer === 7 && tc >= 3) { move = "DOUBLE"; deviationText = "I18 #12: Double 9 vs 7 (TC ≥ +3)"; }
                else { move = (dealer >= 3 && dealer <= 6) ? "DOUBLE" : "HIT"; }
            }
            else if (hand === '8' || hand === '8 or less') { move = "HIT"; }
            else if (hand.startsWith('soft')) {
                move = (dealer >= 5 && dealer <= 6) ? "DOUBLE" : "HIT";
            }

            let recBox = document.getElementById('rec-box');
            let recValue = document.getElementById('rec-value');
            let badge = document.getElementById('deviation-badge');

            recValue.innerText = move;
            if (isSplit || move === 'SPLIT') {
                recBox.className = "recommendation-box split-rec";
            } else {
                recBox.className = "recommendation-box";
            }

            if (deviationText) {
                badge.style.display = 'inline-block';
                badge.innerText = `★ ${deviationText}`;
            } else {
                badge.style.display = 'none';
            }
        }

        function undoAction() {
            if (historyStack.length === 0) return;
            let last = historyStack.pop();
            runningTotal = last.runningTotal;
            totalCardsDealt = last.totalCardsDealt;
            totalDecksInShoe = last.totalDecksInShoe;
            currentSeat = last.currentSeat;
            document.getElementById('history-log').innerText = last.logText;
            document.querySelectorAll('.seat-btn').forEach(btn => {
                btn.classList.toggle('active', btn.innerText === currentSeat);
            });
            document.getElementById('active-seat').innerText = currentSeat;
            updateDisplay();
        }

        function resetShoe() {
            if (confirm("Reset shoe and all counts?")) {
                saveState();
                runningTotal = 0;
                totalCardsDealt = 0;
                totalDecksInShoe = 8;
                updateLog("Shoe & Seats Fully Reset");
                updateDisplay();
            }
        }

        function toggleModal(show) {
            document.getElementById('modal-overlay').style.display = show ? 'flex' : 'none';
        }

        updateDisplay();
